class _AdminVisitorsScreenState extends State<AdminVisitorsScreen> {
  List<dynamic> _visitors = [];
  bool _isLoading = true;

  @override
  void initState() {
    super.initState();
    _fetchVisitors();
  }

  Future<void> _fetchVisitors() async {
    try {
      final societyId = await AuthService.getSocietyId();
      final visitors = await VisitorService.getSocietyVisitors(societyId);
      if (mounted) {
        setState(() {
          _visitors = visitors;
          _isLoading = false;
        });
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Error: $e'), backgroundColor: Colors.red));
        setState(() => _isLoading = false);
      }
    }
  }

  Future<void> _handleManualEntry(String visitorId) async {
    try {
      await VisitorService.guardManualEntry(visitorId, 'Manual entry by Admin');
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('✅ Manual Entry Approved'), backgroundColor: Colors.green));
      _fetchVisitors(); // Refresh
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Error: $e'), backgroundColor: Colors.red));
    }
  }

  Future<void> _handleMarkExit(String visitorId) async {
    try {
      await VisitorService.guardMarkExit(visitorId);
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('✅ Visitor Exit Marked'), backgroundColor: Colors.green));
      _fetchVisitors(); // Refresh
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Error: $e'), backgroundColor: Colors.red));
    }
  }

  Color _getStatusColor(String status) {
    switch (status.toLowerCase()) {
      case 'pending': return Colors.orange;
      case 'inside': return Colors.green;
      case 'exited': return Colors.grey;
      default: return Colors.blue;
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Society Visitors'),
        backgroundColor: Colors.blue,
        foregroundColor: Colors.white,
      ),
      body: _isLoading
          ? const Center(child: CircularProgressIndicator())
          : RefreshIndicator(
              onRefresh: _fetchVisitors,
              child: ListView.builder(
                itemCount: _visitors.length,
                itemBuilder: (context, index) {
                  final v = _visitors[index];
                  final status = v['status'] ?? 'Unknown';

                  return Card(
                    margin: const EdgeInsets.symmetric(horizontal: 12, vertical: 6),
                    child: Padding(
                      padding: const EdgeInsets.all(12.0),
                      child: Column(
                        crossAxisAlignment: CrossAxisAlignment.start,
                        children: [
                          Row(
                            mainAxisAlignment: MainAxisAlignment.spaceBetween,
                            children: [
                              Expanded(
                                child: Text(
                                  v['visitorName'] ?? 'Unknown',
                                  style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 16),
                                ),
                              ),
                              Chip(
                                label: Text(status),
                                backgroundColor: _getStatusColor(status).withOpacity(0.2),
                                labelStyle: TextStyle(color: _getStatusColor(status), fontWeight: FontWeight.bold),
                              ),
                            ],
                          ),
                          const SizedBox(height: 4),
                          Text('Mobile: ${v['visitorMobile'] ?? 'N/A'}'),
                          Text('Flat: ${v['flatId']?.toString().substring(0, 8) ?? 'N/A'}...'),
                          const SizedBox(height: 8),
                          Row(
                            mainAxisAlignment: MainAxisAlignment.end,
                            children: [
                              if (status == 'Pending')
                                ElevatedButton.icon(
                                  onPressed: () => _handleManualEntry(v['id']),
                                  icon: const Icon(Icons.login, size: 16),
                                  label: const Text('Manual Entry'),
                                  style: ElevatedButton.styleFrom(backgroundColor: Colors.orange, foregroundColor: Colors.white),
                                ),
                              if (status == 'Inside')
                                Padding(
                                  padding: const EdgeInsets.only(left: 8.0),
                                  child: ElevatedButton.icon(
                                    onPressed: () => _handleMarkExit(v['id']),
                                    icon: const Icon(Icons.logout, size: 16),
                                    label: const Text('Mark Exit'),
                                    style: ElevatedButton.styleFrom(backgroundColor: Colors.red, foregroundColor: Colors.white),
                                  ),
                                ),
                            ],
                          )
                        ],
                      ),
                    ),
                  );
                },
              ),
            ),
    );
  }

                  child: Padding(

                    child: Text(
                      'Failed to load amenities.\n\nError: $_errorMessage', 
                      textAlign: TextAlign.center,
                      style: const TextStyle(color: Colors.red, fontSize: 14)
                    ),
                  ),
                )
              : _amenities.isEmpty
                  ? const Center(
                      child: Text(
                        'No amenities configured for this society yet.',

                      )
                    )

                        final a = _amenities[index];
                        return Card(
                          margin: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
                          child: ListTile(
                            leading: const Icon(Icons.pool, color: Colors.indigo, size: 40),
                            title: Text(a['name'] ?? 'Amenity', style: const TextStyle(fontWeight: FontWeight.bold)),
